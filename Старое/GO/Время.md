Чтобы найти раздницу между временем Duration то нам нужен метод Sub
```GO
start := time.Date(1961, 4, 12, 6, 7, 0, 0, time.UTC)
    // дата и время посадки: 1961 год, 12 апреля, 7 часов 55 минут
    landing := time.Date(1961, 4, 12, 7, 55, 0, 0, time.UTC)

    // вычисляем интервал
    duration := landing.Sub(start)
```

```Go
func whatTime(friend string) time.Time {
	var friends = map[string]string{
		"Серёга": "Омск",
		"Соня":   "Москва",
		"Дима":   "Екатеринбург",
		"Алина":  "Новосибирск",
		"Егор":   "Калининград",
	}
	var offsetUTC = map[string]int{
		"Санкт-Петербург": 3,
		"Москва":          3,
		"Самара":          4,
		"Новосибирск":     7,
		"Екатеринбург":    5,
		"Казань":          3,
		"Омск":            6,
		"Хабаровск":       10,
		"Пермь":           5,
		"Краснодар":       3,
		"Калининград":     2,
	}
	city := friends[friend]
	duration := time.Now().UTC()
	return duration.Add(time.Duration(offsetUTC[city]) * time.Hour)
}

func main() {
	for _, friend := range []string{"Соня", "Дима", "Егор"} {
		t := whatTime(friend)
		fmt.Printf("%s: %d ч.\n", friend, t.Hour())
	}
}
```
