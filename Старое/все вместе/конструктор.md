конструктор и деконструктор 

конструктор срабатывает когда вызываиться класс или создаеться обьект класса
class SimpleTimer
{
public:
	SimpleTimer()
	{
		start = std::chrono::high_resolution_clock::now();
	}
	~SimpleTimer()
	{
		end = std::chrono::high_resolution_clock::now();
		std::chrono::duration<float>duration = end - start;
		std::cout << "Время цыкла " << duration.count() << " секунд" << std::endl;
	}
private:
	std::chrono::time_point<std::chrono::steady_clock>start, end;

};

деконструктор когда выходит из метода в котором он создавался
