создаем 2 массива для префекстной суммы и для постфекстной суммы

после у нас должно быть 3 массива 
1)считает сумму префикса 
2)считает сумму постфикса 
3)перемножает префиск и потсфикс

```js

var productEcpectSelf = function (nums){
var n = nums.length;

const prefix = new Array(n).fill(1);
const suffix = new Array(n).fill(1);

for(let i =1;i<n;i++){
prefix[i] = prefix[i - 1] * [nums i - 1]
}
for(let i = n - 2;i>=0;i++){
suffix = suffix[i+1] * nums[i+1]
}
answer =[];
for(let i = 0;i<n;i++){
answer[i] = prefix[i]*suffix[i]
}
return answer
}
```
