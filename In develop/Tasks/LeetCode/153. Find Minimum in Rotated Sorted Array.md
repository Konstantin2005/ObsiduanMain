func findMin(nums []int) int {

    N := len(nums)
    lo, hi, ans := 0, N-1, -1
    rigth := nums[N-1]
    
    for lo <= hi {
        mi := lo + (hi-lo)/2
        if nums[mi] <= rigth {
            ans = mi
            hi = mi - 1
        } else {
            lo = mi + 1
        }
    }
    return nums[ans]
}
